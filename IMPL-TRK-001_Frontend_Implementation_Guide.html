§1

Setup — event_id & filter state 初始化

1.1 event_id — generate on every push

Every dataLayer.push() must include a unique event_id. Generate it with crypto.randomUUID(). Do NOT reuse across events.

// Helper — call once per push
const pushEvent = (payload) => {
  dataLayer.push({
    event_id: crypto.randomUUID(),
    ...payload
  });
};

1.2 Filter state — maintain per page

Pages with filters require a filter state object. Initialise on page load, update on every Tab/filter interaction. Pass the current state into every content_impression and content_click.

// /giveaway filter state

let filterState = { filter_audience: "all" };

// Update on Tab click
function onAudienceTabClick(value) {
  filterState.filter_audience = value;
  pushEvent({ event: "button_click", page_type: "listing",
    component_type: "tab", button_action_type: "apply_filter",
    filter_audience: value });
}

// Filter state — maintain per page, pass into every impression/click

// Each page maintains its own filter state object.
// Update on every Tab/filter interaction.
// Always read the CURRENT value when pushing content_impression or content_click.

// /giveaway
let filterAudience = "all";  // update on Tab click

// /earn
let tabValue = "all";  // update on Tab switch

// /connect (two independent dimensions)
let filterGroupType  = "all";        // update on top-level Tab click
let filterLifeStage  = "all_stages"; // update on Birth Club dropdown click

// /guide — flashcard section
let flashcardLifeStage = "trimester_1";  // default, update on top Tab click

// /guide — article section (two independent dimensions)
let articleLifeStage       = "all_stages";  // update on article life_stage dropdown
let articleContentCategory = "all";         // update on article category filter
§2

content_impression — two modes 两种触发模式

Phase 1 note / 当前阶段说明： Viewport-based impression tracking (IntersectionObserver) is not yet implemented. Currently all items are pushed when the list renders. This will be upgraded to viewport-based in Phase 2.当前阶段未实现视口检测（IntersectionObserver）。现在是列表渲染完成后立即 push 所有 item。Phase 2 升级为真实视口曝光。

Mode A — Banner & recommendation_widget: one push per item

Position analysis is important here. Push individually, each with a unique event_id and impression_position at root level.

// Each card fires separately — e.g. 3 banner cards = 3 pushes
dataLayer.push({ event_id: crypto.randomUUID(), event: "content_impression",
  component_type: "banner", content_id: "cmp_001", content_type: "lead_gen",
  impression_position: 1 });

dataLayer.push({ event_id: crypto.randomUUID(), event: "content_impression",
  component_type: "banner", content_id: "cmp_002", content_type: "lead_gen",
  impression_position: 2 });

Mode B — Listing page lists: one push per page load with items array

Position analysis is less critical here (no viewport detection). Push once per load with all items in an array. On Load More, push again with the next batch — impression_position must continue from previous page.

// First load — 10 items, positions 1-10
dataLayer.push({
  event_id: crypto.randomUUID(),  // one ID for this batch
  event: "content_impression", page_type: "listing",
  component_type: "campaign_list",
  filter_audience: "all",  // filter fields stay at root level
  items: [
    { content_id: "cmp_001", content_type: "lead_gen", impression_position: 1 },
    { content_id: "cmp_002", content_type: "lead_gen", impression_position: 2 },
    // ... up to cmp_010, impression_position: 10
  ]
});

// User clicks Load More — push button_click first
dataLayer.push({
  event_id: crypto.randomUUID(),
  event: "button_click", page_type: "listing",
  component_type: "campaign_list",
  button_action_type: "load_more",
  current_page: 1, loaded_count: 10
});

// Next batch loads — positions CONTINUE from 11, do NOT reset to 1
dataLayer.push({
  event_id: crypto.randomUUID(),
  event: "content_impression", page_type: "listing",
  component_type: "campaign_list",
  filter_audience: "all",
  items: [
    { content_id: "cmp_011", content_type: "lead_gen", impression_position: 11 },
    { content_id: "cmp_012", content_type: "lead_gen", impression_position: 12 },
    // ... up to impression_position: 20
  ]
});

// On filter change — impression_position RESETS to 1

BQ — how to analyse search conversion

-- Did users who clicked a search result eventually convert?
-- Join search_result_click with purchase via session_id
SELECT
  src.search_topic,
  src.clicked_content_id,
  COUNT(DISTINCT src.session_id) AS clicked_sessions,
  COUNT(DISTINCT p.session_id)   AS converted_sessions
FROM events src
LEFT JOIN events p
  ON src.session_id = p.session_id
  AND p.event_name = 'purchase'
  AND p.event_timestamp > src.event_timestamp
WHERE src.event_name = 'search_result_click'
GROUP BY 1, 2

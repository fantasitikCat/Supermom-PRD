📋 form_submit
Frontend Push前端 Push ✦ Conversion转化
When to fire
触发时机
Lead form API returns 200 success
表单提交 API 返回 200 成功后
When NOT to fire
不触发的情况
On button click, or if API returns error
按钮点击时，或 API 报错时
🚫
Critical: Fire ONLY inside the API success callback. If the API fails, do NOT fire this event.
关键：必须在 API 成功回调里触发，API 失败时绝对不能触发。
async function handleFormSubmit() {
  try {
    const response = await api.submitLeadForm(formData);

    if (response.status === 200) {
      // ✅ Fire AFTER API success
      window.dataLayer = window.dataLayer || [];
      dataLayer.push({ 'event': 'form_submit' });

      showResultPage(response.data.assigned_type);
    }
  } catch (error) {
    // ❌ Do NOT push anything here
    showErrorMessage(error);
  }
}
ℹ️
No extra parameters needed. GA4 automatically reads UTM params from the page URL.
不需要额外参数，GA4 会自动从页面 URL 读取 UTM 参数。

function onResultPageReady(assignedType) {
  if (!assignedType) {
    console.error('assigned_type missing');
    return;
  }

  window.dataLayer = window.dataLayer || [];
  dataLayer.push({
    'event': 'result_view',
    'assigned_type': assignedType  // e.g. 'the_informed_mom'
  });

  setCTAHref(assignedType); // build CTA URL (see cta_click section)
}

function setCTAHref(assignedType) {
  const p = new URLSearchParams(window.location.search);

  const params = new URLSearchParams({
    utm_source:   p.get('utm_source')   || '',
    utm_medium:   p.get('utm_medium')   || '',
    utm_campaign: p.get('utm_campaign') || '2026_sg_mom_quiz',
    utm_content:  p.get('utm_content')  || '',  // pass through unchanged
    quiz_type:    assignedType                     // business attribution param
  });

  const base = `https://supermom.com/3.0/${assignedType}`;
  document.getElementById('cta-btn').href = `${base}?${params}`;
}
